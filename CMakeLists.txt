function(soundshare_target target)
  target_include_directories(${target} PUBLIC ${PROJECT_SOURCE_DIR}/include)
  target_compile_definitions(${target} PUBLIC SOUNDSHARE_VERSION="${PROJECT_VERSION}")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc /Zc:__cplusplus /wd4324)
    target_compile_definitions(${target} PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(SOUNDSHARE_SANITIZE)
    if(MSVC)
      target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
      target_link_options(${target} PRIVATE /DEBUG)
    else()
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endfunction()

if(SOUNDSHARE_BUILD_WINDOWS)
  if(NOT WIN32)
    message(FATAL_ERROR "Windows backend requires Windows")
  endif()
  # SDK 19041 ships a pre-C++20 coroutine projection. Generate a modern, pinned
  # projection rather than editing SDK headers or lowering our language standard.
  FetchContent_Declare(cppwinrt
    URL https://api.nuget.org/v3-flatcontainer/microsoft.windows.cppwinrt/2.0.240405.15/microsoft.windows.cppwinrt.2.0.240405.15.nupkg
    DOWNLOAD_NAME cppwinrt.zip
    URL_HASH SHA256=e889007b5d9235931e7340ddf737d2c346eebdd23c619f1f4f2426a2aae47180)
  FetchContent_GetProperties(cppwinrt)
  if(NOT cppwinrt_POPULATED)
    FetchContent_Populate(cppwinrt)
  endif()
  get_filename_component(SOUNDSHARE_KITS_ROOT
    "[HKEY_LOCAL_MACHINE\\SOFTWARE\\Microsoft\\Windows Kits\\Installed Roots;KitsRoot10]" ABSOLUTE)
  set(SOUNDSHARE_WINMD "${SOUNDSHARE_KITS_ROOT}/UnionMetadata/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/Windows.winmd"
    CACHE FILEPATH "Windows SDK union metadata used to generate C++/WinRT")
  if(NOT EXISTS "${SOUNDSHARE_WINMD}")
    message(FATAL_ERROR "Set SOUNDSHARE_WINMD to your Windows SDK Windows.winmd")
  endif()
  set(SOUNDSHARE_PROJECTION "${CMAKE_BINARY_DIR}/generated/cppwinrt")
  if(NOT EXISTS "${SOUNDSHARE_PROJECTION}/winrt/base.h")
    execute_process(COMMAND "${cppwinrt_SOURCE_DIR}/bin/cppwinrt.exe" -input "${SOUNDSHARE_WINMD}" -output "${SOUNDSHARE_PROJECTION}"
      RESULT_VARIABLE projection_result)
    if(NOT projection_result EQUAL 0)
      message(FATAL_ERROR "C++/WinRT projection generation failed")
    endif()
  endif()
  add_library(soundshare_backend STATIC src/windows/platform.cpp src/windows/receiver.cpp
    src/windows/audio_engine.cpp src/backend/control.cpp src/windows/pipe.cpp)
  soundshare_target(soundshare_backend)
  target_include_directories(soundshare_backend BEFORE PUBLIC "${SOUNDSHARE_PROJECTION}")
  if(MSVC AND MSVC_VERSION LESS 1930)
    # MSVC 19.29 has an ICE with C++/WinRT's consteval slim_source_location.
    # This supported switch only disables extra exception source-location metadata.
    target_compile_definitions(soundshare_backend PUBLIC WINRT_NO_SOURCE_LOCATION)
  endif()
  target_link_libraries(soundshare_backend PUBLIC soundshare_core nlohmann_json::nlohmann_json
    ole32 uuid avrt propsys runtimeobject windowsapp advapi32)
  add_executable(SoundShare.Worker apps/worker.cpp)
  soundshare_target(SoundShare.Worker)
  target_link_libraries(SoundShare.Worker PRIVATE soundshare_backend)
  add_executable(SoundShare.Ctl apps/ctl.cpp)
  soundshare_target(SoundShare.Ctl)
  target_link_libraries(SoundShare.Ctl PRIVATE soundshare_backend)
  add_library(soundshare_desktop_model STATIC src/desktop/model.cpp)
  soundshare_target(soundshare_desktop_model)
  target_link_libraries(soundshare_desktop_model PUBLIC nlohmann_json::nlohmann_json)
  enable_language(RC)
  add_executable(SoundShare.Desktop WIN32 apps/desktop.cpp apps/desktop.rc src/desktop/client.cpp)
  soundshare_target(SoundShare.Desktop)
  target_include_directories(SoundShare.Desktop PRIVATE ${PROJECT_SOURCE_DIR}/apps)
  target_link_libraries(SoundShare.Desktop PRIVATE soundshare_desktop_model soundshare_backend
    comctl32 shell32 gdi32 user32)
  if(MSVC)
    target_link_options(SoundShare.Desktop PRIVATE /MANIFEST:NO)
  endif()
  install(TARGETS SoundShare.Desktop SoundShare.Worker SoundShare.Ctl RUNTIME DESTINATION bin)
  install(FILES README.md README.vi.md THIRD_PARTY.md IMPLEMENTATION_PLAN.vi.md FEATURES.vi.md
    TEST_PLAN.vi.md WINDOWS_FEASIBILITY.vi.md DESTINATION .)
  install(FILES docs/PROTOCOL.md docs/BACKEND_STATUS.vi.md docs/DESKTOP_STATUS.vi.md docs/AUDIT_0.2.1.vi.md docs/PREPUBLISH_TESTS.vi.md DESTINATION docs)
  install(DIRECTORY third_party/licenses/ DESTINATION licenses)
  install(DIRECTORY examples/ DESTINATION examples)
endif()
